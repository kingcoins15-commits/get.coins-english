    function getCoinScaleFactor(coinsParam) {
      let coins = coinsParam !== undefined ? parseInt(coinsParam) : parseInt(selectedCoins);
      if (isNaN(coins) || coins <= 0) coins = 30;

      const clamped = Math.max(15, Math.min(coins, 10000));
      const logRatio = (Math.log10(clamped) - Math.log10(30)) / (Math.log10(3500) - Math.log10(30));
      return Math.max(0.3, Math.min(1.4, 0.3 + logRatio * 0.9));

    class GoldGreenSmokeParticle {
      constructor(spawnX, spawnY, isBurst = false, coinValue = null, forcedSide = null) {
        this.reset(spawnX, spawnY, isBurst, coinValue, forcedSide);
      }

      reset(spawnX, spawnY, isBurst = false, coinValue = null, forcedSide = null) {
        const factor = getCoinScaleFactor(coinValue);
        const sides = ['left', 'right', 'top', 'bottom'];
        this.side = forcedSide || sides[Math.floor(Math.random() * sides.length)];

        if (spawnX !== undefined && spawnY !== undefined) {
          this.x = spawnX;
          this.y = spawnY;
        } else {
          if (this.side === 'left') {
            this.x = -10 + Math.random() * 15;
            this.y = Math.random() * smokeHeight;
          } else if (this.side === 'right') {
            this.x = smokeWidth + 10 - Math.random() * 15;
            this.y = Math.random() * smokeHeight;
          } else if (this.side === 'top') {
            this.x = Math.random() * smokeWidth;
            this.y = -10 + Math.random() * 15;
          } else {
            this.x = Math.random() * smokeWidth;
            this.y = smokeHeight + 5 - Math.random() * 15;
          }
        }

        const speedPower = (isBurst ? 1.3 : 1.0) * (0.8 + factor * 0.4);

        if (this.side === 'left') {
          this.vx = (0.4 + Math.random() * 0.6) * speedPower;
          this.vy = (Math.random() - 0.5) * 0.6;
        } else if (this.side === 'right') {
          this.vx = -(0.4 + Math.random() * 0.6) * speedPower;
          this.vy = (Math.random() - 0.5) * 0.6;
        } else if (this.side === 'top') {
          this.vx = (Math.random() - 0.5) * 0.7;
          this.vy = (0.35 + Math.random() * 0.5) * speedPower;
        } else {
          this.vx = (Math.random() - 0.5) * 0.8;
          this.vy = -(0.45 + Math.random() * 0.55) * speedPower;
        }

        this.radius = (14 + Math.random() * 14) * (0.85 + factor * 0.35);
        this.growthRate = (0.24 + Math.random() * 0.28) * (0.8 + factor * 0.5);
        this.maxRadius = this.radius * (2.2 + Math.random() * 1.5);

        const baseLife = 45 + factor * 45;
        this.maxLife = baseLife + Math.random() * 30;
        this.life = 0;

        this.alphaMax = Math.min(0.38, (0.20 + Math.random() * 0.12) * (0.8 + factor * 0.4));
        this.wobble = Math.random() * Math.PI * 2;
        this.wobbleSpeed = 0.02 + Math.random() * 0.025;
        this.wobbleWidth = 0.6 + factor * 0.5;

        this.colorType = Math.floor(Math.random() * 3);
      }

      update() {
        this.life++;
        this.wobble += this.wobbleSpeed;

        if (this.side === 'left' || this.side === 'right') {
          this.x += this.vx;
          this.y += this.vy + Math.sin(this.wobble) * this.wobbleWidth;
        } else {
          this.x += this.vx + Math.sin(this.wobble) * this.wobbleWidth;
          this.y += this.vy;
        }

        if (this.radius < this.maxRadius) {
          this.radius += this.growthRate;
        }

        return this.life <= this.maxLife;
      }

      draw(ctx) {
        const progress = this.life / this.maxLife;
        let alpha = this.alphaMax;
        if (progress < 0.2) {
          alpha = (progress / 0.2) * this.alphaMax;
        } else {
          alpha = (1 - progress) * this.alphaMax;
        }

        const grad = ctx.createRadialGradient(this.x, this.y, 0, this.x, this.y, this.radius);

        if (this.colorType === 0) {
          grad.addColorStop(0, `rgba(245, 158, 11, ${alpha * 0.95})`);
          grad.addColorStop(0.35, `rgba(234, 179, 8, ${alpha * 0.7})`);
          grad.addColorStop(0.7, `rgba(163, 230, 53, ${alpha * 0.3})`);
          grad.addColorStop(1, 'rgba(52, 211, 153, 0)');
        } else if (this.colorType === 1) {
          grad.addColorStop(0, `rgba(16, 185, 129, ${alpha * 0.95})`);
          grad.addColorStop(0.35, `rgba(5, 150, 105, ${alpha * 0.7})`);
          grad.addColorStop(0.7, `rgba(52, 211, 153, ${alpha * 0.35})`);
          grad.addColorStop(1, 'rgba(6, 95, 70, 0)');
        } else {
          grad.addColorStop(0, `rgba(234, 179, 8, ${alpha * 0.95})`);
          grad.addColorStop(0.35, `rgba(34, 197, 94, ${alpha * 0.7})`);
          grad.addColorStop(0.7, `rgba(16, 185, 129, ${alpha * 0.35})`);
          grad.addColorStop(1, 'rgba(16, 185, 129, 0)');
        }

        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fill();
      }
    }

    class CoinSparkleParticle {
      constructor(x, y, coinValue = null, forcedSide = null) {
        const factor = getCoinScaleFactor(coinValue);
        const sides = ['left', 'right', 'top', 'bottom'];
        const side = forcedSide || sides[Math.floor(Math.random() * sides.length)];

        if (x !== undefined && y !== undefined) {
          this.x = x;
          this.y = y;
        } else {
          if (side === 'left') {
            this.x = Math.random() * 20;
            this.y = Math.random() * smokeHeight;
          } else if (side === 'right') {
            this.x = smokeWidth - Math.random() * 20;
            this.y = Math.random() * smokeHeight;
          } else if (side === 'top') {
            this.x = Math.random() * smokeWidth;
            this.y = Math.random() * 18;
          } else {
            this.x = Math.random() * smokeWidth;
            this.y = smokeHeight - Math.random() * 20;
          }
        }
        
        this.vx = (Math.random() - 0.5) * (1.6 + factor * 1.3);
        this.vy = (Math.random() - 0.5) * (1.6 + factor * 1.3);
        this.size = (1.8 + Math.random() * 3) * (0.85 + factor * 0.35);
        this.life = 0;
        this.maxLife = 26 + Math.random() * 22;
        this.isGold = Math.random() > 0.45;
        this.rotation = Math.random() * Math.PI;
        this.rotSpeed = (Math.random() - 0.5) * 0.25;
      }

      update() {
        this.life++;
        this.x += this.vx;
        this.y += this.vy;
        this.rotation += this.rotSpeed;
        return this.life <= this.maxLife;
      }

      draw(ctx) {
        const progress = this.life / this.maxLife;
        const alpha = Math.sin(progress * Math.PI);
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate(this.rotation);

        ctx.fillStyle = this.isGold
          ? `rgba(234, 179, 8, ${alpha * 0.95})`
          : `rgba(16, 185, 129, ${alpha * 0.95})`;

        ctx.beginPath();
        const s = this.size;
        ctx.moveTo(0, -s * 1.5);
        ctx.quadraticCurveTo(0, 0, s * 1.5, 0);
        ctx.quadraticCurveTo(0, 0, 0, s * 1.5);
        ctx.quadraticCurveTo(0, 0, -s * 1.5, 0);
        ctx.quadraticCurveTo(0, 0, 0, -s * 1.5);
        ctx.fill();

        ctx.restore();
      }
    }

          sp.draw(smokeCtx);
        } else {
          coinSparkles.splice(j, 1);
        }
      }

      requestAnimationFrame(animateSmoke);
    }

    function extractYouTubeId(url) {
      if (!url) return null;
      const trimmed = url.trim();
      const regExp = /(?:youtu\.be\/|youtube\.com\/(?:embed\/|v\/|watch\?v=|watch\?.+&v=|shorts\/|live\/))([\w-]{11})/;
      const match = trimmed.match(regExp);
      if (match && match[1]) {
        return match[1];
      }
      if (/^[\w-]{11}$/.test(trimmed)) {
        return trimmed;
      }
      return null;
    }

    let detectTimer = null;
    let requestId = 0;

    function normalizeUsername(str) {
      return String(str || '').trim().replace(/^https?:\/\/(?:www\.|m\.)?tiktok\.com\/@?/i, '').replace(/^@/, '').split(/[/?#]/)[0];
    }

    function fetchTikTokProfile(username) {
      const cleanUser = normalizeUsername(username);
      if (!cleanUser || cleanUser === INITIAL_DEFAULT_USERNAME) return;

      const currentReq = ++requestId;
      searchSpinner.classList.add('show');

      fetch('https://www.tikwm.com/api/user/info/', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: 'unique_id=' + encodeURIComponent(cleanUser)
      })
      .then(res => res.json())
      .then(json => {
        if (currentReq !== requestId) return;
        searchSpinner.classList.remove('show');

        if (json.code === 0 && json.data) {
          const u = json.data.user || {};
          currentAccount = {
            username: u.uniqueId || cleanUser,
            name: u.nickname || cleanUser,
            avatar: u.avatarLarger || u.avatarThumb || defaultAvatar
          };
          showTargetPreview(currentAccount);
          addSmokeBurst(14, selectedCoins);
        } else {
          currentAccount = { username: cleanUser, name: cleanUser, avatar: defaultAvatar };
          hideTargetPreview();
          lookupHelper.textContent = `Account selected: @${cleanUser}`;
        }
      })
      .catch(() => {
        if (currentReq !== requestId) return;
        searchSpinner.classList.remove('show');
        currentAccount = { username: cleanUser, name: cleanUser, avatar: defaultAvatar };
        hideTargetPreview();
        lookupHelper.textContent = `Account selected: @${cleanUser}`;
      });
    }
